#include <WiFi.h>
#include <HTTPClient.h>
#include <SPI.h>
#include <MFRC522.h>
#include <Wire.h>
#include <Adafruit_GFX.h>
#include <Adafruit_SH110X.h>

// ESP32 GPIO Pins for RC522
#define SS_PIN  5
#define RST_PIN 2

MFRC522 rfid(SS_PIN, RST_PIN);

// OLED Definitions
#define SCREEN_WIDTH 128
#define SCREEN_HEIGHT 64
#define OLED_RESET -1

Adafruit_SH1106G display(SCREEN_WIDTH, SCREEN_HEIGHT, &Wire, OLED_RESET);

// Network & Database Credentials
const char* ssid = "MJHAY";
const char* password = "123456789";

const char* supabase_base = "https://bltyimpbtazngmfjprxy.supabase.co/rest/v1";
const char* supabase_key = "sb_publishable_t9g0Pb40MrHgWYtrx2kC_Q_ufByRXtL";

// Room details
String roomId = "CS-101";
String currentStatus = "Unknown";   // mirrors rooms.status in the database

const unsigned long HEARTBEAT_MS = 60000;
unsigned long lastHeartbeat = 0;

// ---------- helpers ----------

void addSupabaseHeaders(HTTPClient &http) {
  http.addHeader("apikey", supabase_key);
  http.addHeader("Authorization", String("Bearer ") + String(supabase_key));
}

// Reads the authoritative room status from the database
void fetchRoomStatus() {
  if (WiFi.status() != WL_CONNECTED) return;

  HTTPClient http;
  http.setTimeout(8000);
  http.begin(String(supabase_base) + "/rooms?id=eq." + roomId + "&select=status");
  addSupabaseHeaders(http);

  int code = http.GET();
  if (code == 200) {
    String body = http.getString();
    if (body.indexOf("\"Occupied\"") >= 0) currentStatus = "Occupied";
    else if (body.indexOf("\"Vacant\"") >= 0) currentStatus = "Vacant";
  } else {
    Serial.println("Status fetch failed: " + String(code));
  }
  http.end();
}

// Tells the dashboard this device is alive
void sendHeartbeat() {
  if (WiFi.status() != WL_CONNECTED) {
    WiFi.reconnect();
    return;
  }

  HTTPClient http;
  http.setTimeout(8000);
  http.begin(String(supabase_base) + "/device_heartbeats");
  http.addHeader("Content-Type", "application/json");
  addSupabaseHeaders(http);
  http.addHeader("Prefer", "resolution=merge-duplicates,return=minimal");

  String payload = "{\"room_id\":\"" + roomId + "\",\"rssi\":" + String(WiFi.RSSI()) + "}";
  int code = http.POST(payload);
  Serial.println("Heartbeat: " + String(code));
  http.end();
}

// Helper function to show the standby screen
void showReadyScreen() {
  display.clearDisplay();

  display.setTextSize(1);
  display.setTextColor(SH110X_WHITE);

  display.setCursor(0, 10);

  display.println("Scanner Ready");
  display.println("Room: " + roomId);
  display.println("Status: " + currentStatus);
  display.println("");
  display.println("Tap Card to toggle");

  display.display();
}

// ---------- setup ----------

void setup() {
  Serial.begin(115200);

  // Initialize OLED
  if (!display.begin(0x3C, true)) {
    Serial.println(F("SH1106 allocation failed"));
    for (;;) ;
  }

  // Show Boot Screen
  display.clearDisplay();
  display.setTextSize(1);
  display.setTextColor(SH110X_WHITE);
  display.setCursor(0, 20);
  display.println("System Booting...");
  display.display();

  // Initialize SPI bus and RFID reader
  SPI.begin();
  rfid.PCD_Init();
  rfid.PCD_DumpVersionToSerial();
  Serial.println("RFID Reader Ready. Tap a card.");

  // Connect to Wi-Fi
  WiFi.begin(ssid, password);
  Serial.print("Connecting to Wi-Fi");

  display.clearDisplay();
  display.setCursor(0, 20);
  display.println("Connecting to Wi-Fi...");
  display.display();

  while (WiFi.status() != WL_CONNECTED) {
    delay(500);
    Serial.print(".");
  }
  Serial.println("\nWi-Fi Connected!");

  fetchRoomStatus();
  sendHeartbeat();
  lastHeartbeat = millis();

  showReadyScreen();
}

// ---------- loop ----------

void loop() {
  // Periodic heartbeat (runs even when nobody taps)
  if (millis() - lastHeartbeat >= HEARTBEAT_MS) {
    lastHeartbeat = millis();
    sendHeartbeat();
  }

  // Check if a new card is present and can be read
  if (!rfid.PICC_IsNewCardPresent() || !rfid.PICC_ReadCardSerial()) {
    delay(50);
    return;
  }

  // Extract the UID from the card
  String cardUID = "";
  for (byte i = 0; i < rfid.uid.size; i++) {
    cardUID += String(rfid.uid.uidByte[i] < 0x10 ? "0" : "");
    cardUID += String(rfid.uid.uidByte[i], HEX);
  }
  cardUID.toUpperCase();

  // Stop reading this specific tap to prevent rapid-fire triggering
  rfid.PICC_HaltA();

  // Update OLED to show processing
  display.clearDisplay();
  display.setTextSize(1);
  display.setCursor(0, 10);
  display.println("Processing...");
  display.println("UID: " + cardUID);
  display.display();

  Serial.println("-------------------------");
  Serial.println("Card UID: " + cardUID);

  if (WiFi.status() == WL_CONNECTED) {
    HTTPClient http;
    http.setTimeout(8000);
    http.begin(String(supabase_base) + "/rfid_logs");

    http.addHeader("Content-Type", "application/json");
    addSupabaseHeaders(http);
    http.addHeader("Prefer", "return=minimal");

    // The database flips the status; the value sent here is only a fallback
    String fallback = (currentStatus == "Occupied") ? "Vacant" : "Occupied";
    String jsonPayload =
      "{\"card_uid\":\"" + cardUID +
      "\", \"room_id\":\"" + roomId +
      "\", \"status\":\"" + fallback + "\"}";

    int httpResponseCode = http.POST(jsonPayload);

    if (httpResponseCode == 201) {
      Serial.println("Success! Tap logged.");
      http.end();

      // Read back what the database actually decided
      fetchRoomStatus();
      Serial.println("Room status is now: " + currentStatus);

      display.clearDisplay();
      display.setTextSize(2);
      display.setCursor(0, 10);
      display.println("SUCCESS!");
      display.setTextSize(1);
      display.println("");
      display.println("Set to: " + currentStatus);
      display.display();

    } else {
      Serial.println("Error Code: " + String(httpResponseCode));
      Serial.println("Response Body: " + http.getString());
      http.end();

      display.clearDisplay();
      display.setTextSize(2);
      display.setCursor(0, 10);
      display.println("ERROR");
      display.setTextSize(1);
      display.println("Code: " + String(httpResponseCode));
      display.display();
    }
  } else {
    Serial.println("Wi-Fi disconnected. Update failed.");

    display.clearDisplay();
    display.setTextSize(2);
    display.setCursor(0, 10);
    display.println("OFFLINE");
    display.setTextSize(1);
    display.println("Check Wi-Fi");
    display.display();
  }

  // 3-second cooldown before allowing the next scan
  delay(3000);

  showReadyScreen();
}
